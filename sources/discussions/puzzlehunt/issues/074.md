import hashlib

passwords = [

    "lastwordsbeforearchichoice",
    "thispassword",
    "matrixsumlist",
    "yourlastcommand",
    "secondanswer"
]

hashes = []
for pwd in passwords:
    h = hashlib.sha256(pwd.encode()).digest()
    hashes.append(h)

result = bytearray(32)
for h in hashes:
    for i in range(32):
        result[i] ^= h[i]

print(f"XOR Result: {result.hex()}")
with open('xor_result.txt', 'w') as f:
    f.write(result.hex())
PYTHON_EOF
